# Smart Greenhouse Full-App Shadcn Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (\`- [ ]\`) syntax for tracking.

**Goal:** Replace the rendered Astryx dashboard with the approved shadcn-based Greenhouse Command Deck while preserving all demo interactions, persistence, API boundaries, accessibility, and responsive behavior across all seven application areas.

**Architecture:** Keep \`components/greenhouse-dashboard.tsx\` as a compatibility export and move the active client into focused modules under \`components/greenhouse/\`. Pure presentation selectors in \`lib/greenhouse-presentation.ts\` adapt \`DemoState\` into typed page props; the existing demo store and API/server modules remain the source of truth. shadcn primitives live under \`components/ui/\`, while the official \`dashboard-01\` block is installed once, adapted into greenhouse modules, and then its unused generic scaffold is removed.

**Tech Stack:** Node.js >=22.13.0, Next.js 16.2.6, React 19.2.6, TypeScript 5.9 strict mode, Tailwind CSS 4.2.1, shadcn/ui latest New York/Radix/CSS-variable configuration, Lucide React, Recharts, Vinext, Node test runner.

## Global Constraints

- UI/client-side only: do not change API routes, database schemas, authentication, deployment, CI/CD, or infrastructure.
- Preserve \`lib/greenhouse-demo-store.ts\`, \`lib/dashboard-interactions.ts\` behavior, \`lib/mock-data.ts\` sensor exports used by \`app/api/sensors/route.ts\`, and all server contracts.
- Thai is the rendered interface language; code comments remain English.
- shadcn/ui is the only rendered design system at completion. Remove Astryx imports and packages after parity passes.
- Use the approved light palette: primary \`#25734F\`, hover \`#174B36\`, soft \`#EFF8F2\`, canvas \`#F3F5F3\`, surface \`#FFFFFF\`, border \`#DFE5E1\`, primary text \`#1D2923\`, secondary text \`#4F5C55\`, warning \`#8A5108\`, danger \`#B9413A\`.
- Use \`Noto Sans Thai, Leelawadee UI, Tahoma, Arial, sans-serif\`; keep tabular numerals for telemetry.
- WCAG 2.1 AA, visible focus, semantic status text, reduced-motion support, and 44 by 44 pixel primary touch targets are mandatory.
- Verify 390, 768, 1024, and 1440 pixel widths with no page-level horizontal overflow.
- Because the demo store has no server freshness metadata, mark the client stale five minutes after the last successful load; refresh resets the timer. Browser offline events preserve the last state, mark it stale, and disable device-command entry points with an explicit reason.
- Do not stage \`.agents/\`, \`AGENTS.md\`, \`skills-lock.json\`, \`.vinext/\`, \`.superpowers/\`, or \`tsconfig.tsbuildinfo\` with feature commits.
- The current dirty legacy file \`components/dashboard/dashboard-view.tsx\` may be deleted only in Task 9 after the import graph proves its sole consumer is the unused \`components/smart-greenhouse-app.tsx\`; the cleanup commit keeps it recoverable.

  state: DemoState,
): DashboardViewModel {
  const healthScore = Math.round(
    state.plants.reduce((total, plant) => total + plant.confidence, 0) /
      state.plants.length,
  );
  const activeDevices = state.devices.filter((device) => device.active).length;
  const openAlerts = state.alerts.filter((alert) => !alert.resolved);
  const warningPlant = state.plants.find(
    (plant) => plant.health === "ควรตรวจสอบ",
  );

  return {
    healthScore,
    activeDevices,
    deviceCount: state.devices.length,
    openAlerts: openAlerts.length,
    metrics: [
      {
        id: "health",
        label: "สุขภาพพืช",

export function filterPlants(
  plants: DemoPlant[],
  query: string,
  health: PlantFilter,
): DemoPlant[] {
  const normalized = query.trim().toLocaleLowerCase();
  return plants.filter((plant) => {
    const matchesHealth = health === "all" || plant.health === health;
    const haystack = [plant.name, plant.id, plant.zone]
      .join(" ")
      .toLocaleLowerCase();
    return matchesHealth && (!normalized || haystack.includes(normalized));
  });
}

export function filterAlerts(
  alerts: DemoAlert[],
  filter: AlertFilter,
): DemoAlert[] {
  if (filter === "open") return alerts.filter((alert) => !alert.resolved);
  if (filter === "resolved") return alerts.filter((alert) => alert.resolved);
  return alerts;
}

export function describeSoilMoistureTrend(
  points: SoilMoisturePoint[],
  minTarget: number,
): string {
  const first = points.at(0)?.value ?? 0;
  const current = points.at(-1)?.value ?? 0;
  const direction = current < first ? "ลดลง" : current > first ? "เพิ่มขึ้น" : "คงที่";
  const targetCopy = current < minTarget
    ? "ต่ำกว่าเป้าหมาย " + String(minTarget - current) + "%"
    : "อยู่ในช่วงเป้าหมาย";
  return "ความชื้นดิน" + direction + "เหลือ " + String(current) + "% " + targetCopy;
}

"use client";

import { Search } from "lucide-react";
import { Input } from "@/components/ui/input";
import type { DashboardSearchResult } from "@/lib/dashboard-interactions";

export function GlobalSearch({
  idPrefix,
  value,
  activeIndex,
  results,
  autoFocus,
  onValueChange,
  onActiveIndexChange,
  onChoose,
}: {
  idPrefix: string;
  value: string;
  activeIndex: number;
  results: DashboardSearchResult[];
  autoFocus?: boolean;
  onValueChange: (value: string) => void;
  onActiveIndexChange: (index: number) => void;
  onChoose: (result: DashboardSearchResult) => void;
}) {
  const listId = idPrefix + "-results";
  return (
    <div className="relative w-full">
      <Search className="pointer-events-none absolute left-3 top-1/2 z-10 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
      <Input
        autoFocus={autoFocus}
        className="min-h-11 pl-9"
        role="combobox"
        aria-label="ค้นหาทั่วทั้งโรงเรือน"
        aria-controls={listId}
        aria-expanded={Boolean(value)}
        aria-autocomplete="list"
        aria-activedescendant={results[activeIndex] ? idPrefix + "-option-" + results[activeIndex].key : undefined}
        placeholder="ค้นหาพืช อุปกรณ์ หรือเหตุการณ์"
        value={value}
        onChange={(event) => { onValueChange(event.target.value); onActiveIndexChange(0); }}
        onKeyDown={(event) => {
          if (event.key === "Escape") { onValueChange(""); return; }
          if (!results.length) return;
          if (event.key === "ArrowDown") { event.preventDefault(); onActiveIndexChange((activeIndex + 1) % results.length); }
          if (event.key === "ArrowUp") { event.preventDefault(); onActiveIndexChange((activeIndex - 1 + results.length) % results.length); }
          if (event.key === "Home") { event.preventDefault(); onActiveIndexChange(0); }
          if (event.key === "End") { event.preventDefault(); onActiveIndexChange(results.length - 1); }
          if (event.key === "Enter") { event.preventDefault(); onChoose(results[activeIndex] ?? results[0]); }
        }}
      />
      {value ? (
        <div id={listId} role="listbox" aria-label="ผลการค้นหา" className="absolute inset-x-0 top-[calc(100%+0.5rem)] z-50 max-h-72 overflow-y-auto rounded-lg border bg-popover p-1 shadow-lg">
          {results.length ? results.map((result, index) => (
            <button id={idPrefix + "-option-" + result.key} key={result.key} role="option" aria-selected={index === activeIndex} className="flex min-h-11 w-full flex-col justify-center rounded-md px-3 text-left hover:bg-accent aria-selected:bg-accent" onMouseDown={(event) => event.preventDefault()} onClick={() => onChoose(result)}>
              <strong className="text-sm">{result.label}</strong>
              <small className="text-muted-foreground">{result.detail}</small>
            </button>
          )) : <p className="p-4 text-center text-sm text-muted-foreground">ไม่พบผลลัพธ์</p>}
        </div>
      ) : null}
    </div>
  );
}
